          cocoa-light cocoa-dark plum-light plum-dark"
  default-theme="oat-light"
  initial-mode="system"
  transition='{"enabled":true,"duration":220,"easing":"ease-out"}'
>
  <themvora-select type="family" label="Theme family"></themvora-select>
  <themvora-select type="mode" label="Colour mode"></themvora-select>
  <themvora-toggle></themvora-toggle>
</themvora-provider>

2 · The boot script

<!-- index.html — blocking, so the elements exist at first paint -->
<script src="/themvora-web.js"></script>

// main.ts — the same package, read off the global it exposes
const { getProviderRuntime } = window.ThemvoraWeb;

const provider = document.querySelector<HTMLElement>("#provider");
const runtime = getProviderRuntime(provider);

// Getters, not subscriptions: subscribe, then re-read.
const render = () => {
  const theme = runtime.store.get();
  console.log(theme.name, runtime.selection.getMode());
};

runtime.store.subscribe(render);
render();

3 · A catalog change

// Installs a pack the page did not ship with.
runtime.use({ name: "solar", themes: [solarLight, solarDark] });

// A catalog change that preserves the active theme emits no store
// change — the registry signal is what scopes and selects follow.
runtime.registry.subscribe(() => {
  console.log(runtime.themes.map((theme) => theme.name));
});

// The DOM event carries the same information for page code.
provider.addEventListener("themes-change", (event) => {
  console.log(event.detail.themes);
});
